#pragma once
#include <JuceHeader.h>

namespace Aetheria {

class Envelope {
public:
    Envelope() = default;
    
    void setSampleRate(double sr) { sampleRate = sr; }
    void setParameters(float a, float d, float s, float r) {
        attack = juce::jmax(0.001f, a);
        decay = juce::jmax(0.001f, d);
        sustain = juce::jlimit(0.0f, 1.0f, s);
        release = juce::jmax(0.001f, r);
    }
    
    void noteOn() { state = Attack; counter = 0; }
    void noteOff() { if (state != Idle) { state = Release; counter = 0; releaseStart = level; } }
    void reset() { state = Idle; level = 0.0f; counter = 0; }
    
    bool isActive() const { return state != Idle; }
    bool isReleasing() const { return state == Release; }
    
    float getNextSample() {
        if (state == Idle) return 0.0f;
        
        switch (state) {
            case Attack: {
                int samples = static_cast<int>(attack * sampleRate);
                level = (counter++ >= samples) ? (state = Decay, counter = 0, 1.0f) 
                                               : static_cast<float>(counter) / samples;
                break;
            }
            case Decay: {
                int samples = static_cast<int>(decay * sampleRate);
                level = (counter++ >= samples) ? (state = Sustain, sustain) 
                                               : 1.0f - (static_cast<float>(counter) / samples) * (1.0f - sustain);
                break;
            }
            case Sustain:
                level = sustain;
                break;
            case Release: {
                int samples = static_cast<int>(release * sampleRate);
                if (counter++ >= samples) { level = 0.0f; state = Idle; }
                else level = releaseStart * (1.0f - static_cast<float>(counter) / samples);
                break;
            }
        }
        return level;
    }
    
private:
    enum State { Idle, Attack, Decay, Sustain, Release };
    State state = Idle;
    double sampleRate = 44100.0;
    float level = 0.0f;
    float attack = 0.01f, decay = 0.1f, sustain = 0.7f, release = 0.3f;
    float releaseStart = 0.0f;
    int counter = 0;
};

} // namespace Aetheria

#pragma once
#include <JuceHeader.h>

namespace Aetheria {

class Oscillator {
public:
    void setSampleRate(double sr) { sampleRate = sr; }
    void setFrequency(float freq) { frequency = freq; phaseIncrement = frequency / static_cast<float>(sampleRate); }
    void reset() { phase = 0.0f; }
    
    float getNextSample() {
        float sample = std::sin(phase * juce::MathConstants<float>::twoPi);
        phase += phaseIncrement;
        if (phase >= 1.0f) phase -= 1.0f;
        return sample;
    }
    
private:
    double sampleRate = 44100.0;
    float frequency = 440.0f;
    float phase = 0.0f;
    float phaseIncrement = 0.0f;
};

} // namespace Aetheria

#pragma once
#include <JuceHeader.h>

namespace Aetheria {

class Filter {
public:
    void setSampleRate(double sr) { sampleRate = sr; updateCoefficients(); }
    void setParameters(float cutoff, float res) {
        frequency = juce::jlimit(20.0f, 20000.0f, cutoff);
        resonance = juce::jlimit(0.1f, 10.0f, res);
        updateCoefficients();
    }
    
    void reset() { z1 = z2 = 0.0f; }
    
    float process(float input) {
        float output = a0 * input + a1 * z1 + a2 * z2 - b1 * z1 - b2 * z2;
        z2 = z1;
        z1 = input;
        return output;
    }
    
private:
    void updateCoefficients() {
        float w0 = juce::MathConstants<float>::twoPi * frequency / static_cast<float>(sampleRate);
        float cosw0 = std::cos(w0);
        float alpha = std::sin(w0) / (2.0f * resonance);
        
        float b0 = 1.0f + alpha;
        a0 = ((1.0f - cosw0) / 2.0f) / b0;
        a1 = (1.0f - cosw0) / b0;
        a2 = a0;
        b1 = (-2.0f * cosw0) / b0;
        b2 = (1.0f - alpha) / b0;
    }
    
    double sampleRate = 44100.0;
    float frequency = 1000.0f;
    float resonance = 0.707f;
    float a0 = 0.0f, a1 = 0.0f, a2 = 0.0f, b1 = 0.0f, b2 = 0.0f;
    float z1 = 0.0f, z2 = 0.0f;
};

} // namespace Aetheria
